int ColR(color c) { return (int)(((int)c)&0xFF); }
int ColG(color c) { return (int)((((int)c)>>8)&0xFF); }
int ColB(color c) { return (int)((((int)c)>>16)&0xFF); }
//--- share is the proportion of colour a kept in the mix
color MixColor(color a,color b,double share) {
    if(share<0)share=0; if(share>1)share=1;
    int r=(int)MathRound(ColR(a)*share+ColR(b)*(1.0-share));
    int g=(int)MathRound(ColG(a)*share+ColG(b)*(1.0-share));
    int bl=(int)MathRound(ColB(a)*share+ColB(b)*(1.0-share));
    return (color)((bl<<16)|(g<<8)|r);
}

//--- raw pixel rectangle, used for corner masks and hairlines
void RawRect(string name,int px,int py,int w,int h,color bg,int zorder) {
    string n=DashName(name); if(ObjectFind(0,n)<0) ObjectCreate(0,n,OBJ_RECTANGLE_LABEL,0,0,0);
    ObjectSetInteger(0,n,OBJPROP_CORNER,InpDashboardCorner);
    ObjectSetInteger(0,n,OBJPROP_XDISTANCE,px); ObjectSetInteger(0,n,OBJPROP_YDISTANCE,py);
    ObjectSetInteger(0,n,OBJPROP_XSIZE,MathMax(1,w)); ObjectSetInteger(0,n,OBJPROP_YSIZE,MathMax(1,h));
    ObjectSetInteger(0,n,OBJPROP_BGCOLOR,bg); ObjectSetInteger(0,n,OBJPROP_BORDER_TYPE,BORDER_FLAT); ObjectSetInteger(0,n,OBJPROP_COLOR,bg);
    ObjectSetInteger(0,n,OBJPROP_SELECTABLE,false); ObjectSetInteger(0,n,OBJPROP_HIDDEN,true); ObjectSetInteger(0,n,OBJPROP_ZORDER,zorder);
}
//--- MT5 objects are square, so the corners are masked with the surrounding colour
void RoundCorners(string base,int px,int py,int pw,int ph,color outer,int radius,int zorder) {
    int r=MathMax(0,MathMin(18,radius));
    for(int i=0;i<r;i++) {
        double row_distance=(double)(r-i);
        int len=(int)MathCeil((double)r-MathSqrt((double)(r*r)-row_distance*row_distance));
        RawRect("RC_"+base+"_TL"+IntegerToString(i),px,py+i,len,1,outer,zorder);
        RawRect("RC_"+base+"_TR"+IntegerToString(i),px+pw-len,py+i,len,1,outer,zorder);
        RawRect("RC_"+base+"_BL"+IntegerToString(i),px,py+ph-1-i,len,1,outer,zorder);
        RawRect("RC_"+base+"_BR"+IntegerToString(i),px+pw-len,py+ph-1-i,len,1,outer,zorder);
    }
}

void SetDashboardPanel(string name,int x,int y,int width,int height,color bg,color border,bool draggable=false) {
    string n=DashName(name); bool created=ObjectFind(0,n)<0;
    if(created) ObjectCreate(0,n,OBJ_RECTANGLE_LABEL,0,0,0);
    ObjectSetInteger(0,n,OBJPROP_CORNER,InpDashboardCorner); ObjectSetInteger(0,n,OBJPROP_XDISTANCE,dashboard_x+DS(x)); ObjectSetInteger(0,n,OBJPROP_YDISTANCE,dashboard_y+DS(y));
    ObjectSetInteger(0,n,OBJPROP_XSIZE,DS(width)); ObjectSetInteger(0,n,OBJPROP_YSIZE,DS(height));
    ObjectSetInteger(0,n,OBJPROP_BGCOLOR,bg);
    //--- rectangle labels take their border colour from OBJPROP_COLOR with a flat border
    ObjectSetInteger(0,n,OBJPROP_BORDER_TYPE,BORDER_FLAT); ObjectSetInteger(0,n,OBJPROP_COLOR,border);
    ObjectSetInteger(0,n,OBJPROP_SELECTABLE,draggable);
    if(created) ObjectSetInteger(0,n,OBJPROP_SELECTED,false);
    ObjectSetInteger(0,n,OBJPROP_HIDDEN,!draggable); ObjectSetInteger(0,n,OBJPROP_ZORDER,draggable?2:0);
}

double TodayNetProfit(int &wins,int &losses,int &deals) {
    wins=0; losses=0; deals=0; MqlDateTime dt; TimeToStruct(TimeCurrent(),dt); dt.hour=0;dt.min=0;dt.sec=0;
    if(!HistorySelect(StructToTime(dt),TimeCurrent())) return 0; double total=0;
    for(int i=0;i<HistoryDealsTotal();i++){ ulong t=HistoryDealGetTicket(i); if(t==0 || HistoryDealGetInteger(t,DEAL_MAGIC)!=EA_MAGIC_NUMBER)continue; long e=HistoryDealGetInteger(t,DEAL_ENTRY); if(e!=DEAL_ENTRY_OUT&&e!=DEAL_ENTRY_OUT_BY)continue; double p=HistoryDealGetDouble(t,DEAL_PROFIT)+HistoryDealGetDouble(t,DEAL_COMMISSION)+HistoryDealGetDouble(t,DEAL_SWAP); total+=p;deals++;if(p>=0)wins++;else losses++; }
    return total;
}

color MLLinkColor() {

string CompactMLText(string value,int max_length=70) {
    StringReplace(value,"; "," | ");
    if(StringLen(value)>max_length) return StringSubstr(value,0,max_length-3)+"...";
    return value;
}

void DrawMLPointOfInterest() {
    bool valid=ml_prediction_available &&
        (ml_poi_direction=="BUY" || ml_poi_direction=="SELL") &&
        ml_entry_zone_low>0 && ml_entry_zone_high>=ml_entry_zone_low;
    if(!valid) {
        RemoveMLPointOfInterest();
        return;
    }
    datetime now=TimeCurrent();
    int seconds=MathMax(1,PeriodSeconds(_Period));
    datetime left=now-seconds*20;
    datetime right=now+seconds*6;
    color poi_color=ml_poi_direction=="BUY"?DashGreen():DashRed();
    string zone=ml_poi_prefix+"ZONE";
    if(ObjectFind(0,zone)<0) ObjectCreate(0,zone,OBJ_RECTANGLE,0,left,ml_entry_zone_high,right,ml_entry_zone_low);
    else {
        ObjectMove(0,zone,0,left,ml_entry_zone_high);
        ObjectMove(0,zone,1,right,ml_entry_zone_low);
    }
    ObjectSetInteger(0,zone,OBJPROP_COLOR,poi_color);
    ObjectSetInteger(0,zone,OBJPROP_STYLE,STYLE_DASH);
    ObjectSetInteger(0,zone,OBJPROP_WIDTH,1);
    ObjectSetInteger(0,zone,OBJPROP_FILL,true);
    ObjectSetInteger(0,zone,OBJPROP_BACK,true);
    ObjectSetInteger(0,zone,OBJPROP_SELECTABLE,false);
    ObjectSetInteger(0,zone,OBJPROP_HIDDEN,true);
    string mid=ml_poi_prefix+"MID";
    if(ObjectFind(0,mid)<0) ObjectCreate(0,mid,OBJ_HLINE,0,0,ml_entry_price>0?ml_entry_price:(ml_entry_zone_low+ml_entry_zone_high)*0.5);
    ObjectSetDouble(0,mid,OBJPROP_PRICE,ml_entry_price>0?ml_entry_price:(ml_entry_zone_low+ml_entry_zone_high)*0.5);
    ObjectSetInteger(0,mid,OBJPROP_COLOR,poi_color);
    ObjectSetInteger(0,mid,OBJPROP_STYLE,STYLE_DOT);
    ObjectSetInteger(0,mid,OBJPROP_SELECTABLE,false);
    ObjectSetInteger(0,mid,OBJPROP_HIDDEN,true);
    string label=ml_poi_prefix+"LABEL";
    string caption=StringFormat("NEXA ML POI %s | %s | %s",
        ml_poi_direction,ml_confirmation_status,ml_entry_zone_source);
    if(ObjectFind(0,label)<0) ObjectCreate(0,label,OBJ_TEXT,0,right,ml_entry_zone_high);
    else ObjectMove(0,label,0,right,ml_entry_zone_high);
    ObjectSetString(0,label,OBJPROP_TEXT,caption);
    ObjectSetString(0,label,OBJPROP_FONT,FontMedium());
    ObjectSetInteger(0,label,OBJPROP_FONTSIZE,8);
    ObjectSetInteger(0,label,OBJPROP_COLOR,poi_color);
    ObjectSetInteger(0,label,OBJPROP_ANCHOR,ANCHOR_LEFT_LOWER);
    ObjectSetInteger(0,label,OBJPROP_SELECTABLE,false);
    ObjectSetInteger(0,label,OBJPROP_HIDDEN,true);
    ChartRedraw();
}

bool IsFiniteWeight(double value) { return MathIsValidNumber(value) && value>=InpMinAdaptiveWeight && value<=InpMaxAdaptiveWeight; }
bool ValidateAdaptiveConfiguration() {
    if(!adaptive_config.supplied) { ml_config_state=ML_CONFIG_OFFLINE; return false; }
    if(adaptive_config.generated_at<=0 || adaptive_config.generated_at>TimeCurrent()+300) { ml_config_state=ML_CONFIG_INVALID; return false; }
    if(!MathIsValidNumber(adaptive_config.confidence) || adaptive_config.confidence<0.0 || adaptive_config.confidence>1.0) { ml_config_state=ML_CONFIG_INVALID; return false; }
    if(TimeCurrent()-adaptive_config.generated_at>InpMLConfigTimeoutMinutes*60) { ml_config_state=ML_CONFIG_EXPIRED; return false; }
    for(int i=0;i<INDICATOR_COUNT;i++) if(!IsFiniteWeight(adaptive_config.weight[i])) { ml_config_state=ML_CONFIG_INVALID; return false; }
    ml_config_state=ML_CONFIG_VALID; return true;
}
void ResetToBaseConfiguration() { for(int i=0;i<INDICATOR_COUNT;i++){ tools[i].enabled=tools[i].base_enabled; tools[i].weight=tools[i].base_weight; } }
bool ApplyAdaptiveConfiguration() {
    if(!runtime_adaptive_enabled || !ValidateAdaptiveConfiguration()) { ResetToBaseConfiguration(); return false; }
    for(int i=0;i<INDICATOR_COUNT;i++){ tools[i].enabled=adaptive_config.enabled[i]; tools[i].weight=adaptive_config.weight[i]; }
    return true;
}

int NormalizeHour(int hour) { while(hour<0) hour+=24; while(hour>=24) hour-=24; return hour; }
bool HourInWindow(int hour,int start_hour,int end_hour) { if(start_hour==end_hour) return true; return start_hour<end_hour ? (hour>=start_hour && hour<end_hour) : (hour>=start_hour || hour<end_hour); }
string GetCurrentSession() {
    MqlDateTime dt; TimeToStruct(TimeCurrent(),dt); int utc_hour=NormalizeHour(dt.hour-InpBrokerUtcOffsetHours);
    bool asian=HourInWindow(utc_hour,InpAsianStartUTC,InpAsianEndUTC);
    bool london=HourInWindow(utc_hour,InpLondonStartUTC,InpLondonEndUTC);
    bool ny=HourInWindow(utc_hour,InpNewYorkStartUTC,InpNewYorkEndUTC);
    if(london && ny) return "LONDON_NEW_YORK_OVERLAP"; if(london) return "LONDON"; if(ny) return "NEW_YORK"; if(asian) return "ASIAN"; return "OTHER";
}

string GetVolatilityRegime() {
    double a[]; ArraySetAsSeries(a,true); if(atr_handle==INVALID_HANDLE || CopyBuffer(atr_handle,0,1,21,a)<21) return "UNKNOWN";
    double avg=0; for(int i=1;i<21;i++) avg+=a[i]; avg/=20.0; if(avg<=0) return "UNKNOWN";
    if(a[0]>=avg*InpHighVolatilityRatio) return "HIGH"; if(a[0]<=avg*InpLowVolatilityRatio) return "LOW"; return "NORMAL";
}

int DecisionScopeCode() {
    ulong hash=(ulong)AccountInfoInteger(ACCOUNT_LOGIN);
    for(int i=0;i<StringLen(_Symbol);i++) hash=(hash*31+(ulong)StringGetCharacter(_Symbol,i))%10000;
    hash=(hash*31+(ulong)_Period)%10000;
    return (int)hash;
}
string Base36Code(int value,int width) {
    const string digits="0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    string code="";
    for(int i=0;i<width;i++) {
        code=StringSubstr(digits,value%36,1)+code;
        value/=36;
    }
    return code;
}

bool AcquireDataFileLock(string base) {
    string name=DataFileLockName(base);
    if(!GlobalVariableCheck(name) && !GlobalVariableSet(name,0.0)) return false;
    double token=(double)GetTickCount()+((double)ChartID()/1000000000.0);
    for(int attempt=0;attempt<25;attempt++) {
        double current=GlobalVariableGet(name);
        if(current==0.0 && GlobalVariableSetOnCondition(name,token,0.0)) return true;
        Sleep(10);
    }
    return false;
}
void ReleaseDataFileLock(string base) {
    string name=DataFileLockName(base);
    if(GlobalVariableCheck(name)) GlobalVariableSet(name,0.0);
}
int OpenDataFile(string base) {

    int h=FileOpen(name,flags,',');
    if(h==INVALID_HANDLE && runtime_debug_logs) Print("DATA FILE ERROR ",name," err=",GetLastError());
    return h;
}

void WriteMLRecord(string id,string event_type,string direction,string source,double entry,double sl,double tp,double lots,
                   double bull,double bear,int confirmations,string failed,string result,double net_pl,double r_multiple,bool has_outcome) {
    if(!DataLoggingEnabled()) return;
    //--- one row per decision event, so ticks never duplicate a record
    static string ml_last_key="";
    string key=id+"|"+event_type;
    if(key==ml_last_key) return;
    ml_last_key=key;

    if(!AcquireDataFileLock("ml")) { if(runtime_debug_logs) Print("ML record deferred: log lock unavailable decision=",id); return; }
    int h=OpenDataFile("ml"); if(h==INVALID_HANDLE) { ReleaseDataFileLock("ml"); return; }
    if(FileSize(h)==0) FileWriteString(h,MLHeaderLine()+"\r\n");
    FileSeek(h,0,SEEK_END);

    double spread=(_Point>0)?(SymbolInfoDouble(_Symbol,SYMBOL_ASK)-SymbolInfoDouble(_Symbol,SYMBOL_BID))/_Point:0;
    double total=bull+bear;
    string ratio=(total>0)?DoubleToString(MathMax(bull,bear)/total,4):"NA";
    string adaptive_confidence=(ml_config_state==ML_CONFIG_VALID)?DoubleToString(adaptive_config.confidence,4):"NA";

    string line=MLField(TimeToString(TimeCurrent(),TIME_DATE|TIME_SECONDS))+","+RunContext()+","+_Symbol+","+EnumToString(_Period)+","
        +MLField(GetCurrentSession())+","+MLField(RangingText())+","+MLField(GetMarketRegime())+","+MLField(GetVolatilityRegime())+","+MLNumber(spread,1)+","
        +MLField(id)+","+MLField(event_type)+","+MLField(direction)+","+MLField(source)+","+MLModeText()+","+IntegerToString(confirmations)+","
        +MLNumber(bull,4)+","+MLNumber(bear,4)+","+ratio+","+MLField(failed)+","
        +(entry>0?MLNumber(entry,_Digits):"NA")+","+(sl>0?MLNumber(sl,_Digits):"NA")+","+(tp>0?MLNumber(tp,_Digits):"NA")+","+(lots>0?MLNumber(lots,2):"NA")+","
        +MLStateText()+","+MLField(adaptive_config.source)+","+adaptive_confidence+","
        +MLField(MLLinkStateText())+","+IntegerToString(ml_link_http_code)+","+(ml_link_latency_ms>0?IntegerToString(ml_link_latency_ms):"NA")+","
        +(has_outcome?MLField(result):"NA")+","+(has_outcome?MLNumber(net_pl,2):"NA")+","+(has_outcome?MLNumber(r_multiple,4):"NA");

    for(int i=0;i<INDICATOR_COUNT;i++) {
        bool fresh=(tools[i].last_update>0);
        line+=","+(fresh?MLNumber(tools[i].value):"NA")
             +","+(fresh?MLNumber(tools[i].secondary_value):"NA")
             +","+(fresh?SignalText(tools[i].state):"NA")
             +","+(fresh?MLNumber(tools[i].strength,4):"NA")
             +","+MLNumber(tools[i].weight,2)
             +","+DashboardBool(tools[i].enabled);
    }

    FileWriteString(h,line+"\r\n");
    FileFlush(h); FileClose(h); ReleaseDataFileLock("ml");
}

int CountEAOpenPositions() { int count=0; for(int i=0;i<PositionsTotal();i++){ ulong t=PositionGetTicket(i); if(t>0 && PositionGetInteger(POSITION_MAGIC)==EA_MAGIC_NUMBER && PositionGetString(POSITION_SYMBOL)==_Symbol) count++; } return count; }
bool GetScalpBasketStats(int &count,double &volume,double &average,double &target,double &floating,string &direction) {
    count=0; volume=0; average=0; target=0; floating=0; direction="";
    if(!IsGoldSymbol()) return false;
    double weighted_entry=0;
    for(int i=0;i<PositionsTotal();i++) {
        ulong ticket=PositionGetTicket(i);
        if(ticket==0 || (long)PositionGetInteger(POSITION_MAGIC)!=EA_MAGIC_NUMBER || PositionGetString(POSITION_SYMBOL)!=_Symbol) continue;
        if(PositionGetDouble(POSITION_SL)>0) continue;
        long type=PositionGetInteger(POSITION_TYPE);
        string side=type==POSITION_TYPE_BUY?"BUY":"SELL";
        if(direction!="" && direction!=side) return false;
        direction=side;
        double position_volume=PositionGetDouble(POSITION_VOLUME);
        count++;
        volume+=position_volume;
        weighted_entry+=PositionGetDouble(POSITION_PRICE_OPEN)*position_volume;
        floating+=PositionGetDouble(POSITION_PROFIT)+PositionGetDouble(POSITION_SWAP);
    }
    if(count==0 || volume<=0) { direction=""; return false; }
    average=weighted_entry/volume;
    double stop_buffer=(double)SymbolInfoInteger(_Symbol,SYMBOL_TRADE_STOPS_LEVEL)*_Point+_Point;
    if(direction=="BUY")
        target=MathMax(average+InpScalpTakeProfitPips*ScalpPipSize(),SymbolInfoDouble(_Symbol,SYMBOL_ASK)+stop_buffer);
    else
        target=MathMin(average-InpScalpTakeProfitPips*ScalpPipSize(),SymbolInfoDouble(_Symbol,SYMBOL_BID)-stop_buffer);
    double tick_size=SymbolInfoDouble(_Symbol,SYMBOL_TRADE_TICK_SIZE);
    if(tick_size>0) target=MathRound(target/tick_size)*tick_size;
    target=NormalizeDouble(target,_Digits);
    return true;
}
void UpdateScalpBasketTakeProfit() {
    if(!IsGoldSymbol()) return;
    int count=0; double volume=0,average=0,target=0,floating=0; string direction="";
    if(!GetScalpBasketStats(count,volume,average,target,floating,direction)) { scalp_basket_tp_ready=(count==0); return; }
    scalp_basket_tp_ready=true;
    bool target_changed=false;
    for(int i=0;i<PositionsTotal();i++) {
        ulong ticket=PositionGetTicket(i);
        if(ticket==0 || (long)PositionGetInteger(POSITION_MAGIC)!=EA_MAGIC_NUMBER || PositionGetString(POSITION_SYMBOL)!=_Symbol) continue;
        if(PositionGetDouble(POSITION_SL)>0) continue;
        double current_tp=PositionGetDouble(POSITION_TP);
        double tick_size=SymbolInfoDouble(_Symbol,SYMBOL_TRADE_TICK_SIZE);
        if(MathAbs(current_tp-target)<=MathMax(_Point,tick_size)*0.5) continue;
        if(!trade.PositionModify(ticket,0,target)) {
            scalp_basket_tp_ready=false;
            Print("[NEXA][BASKET][ERROR] TP update failed ticket=",ticket," target=",DoubleToString(target,_Digits)," retcode=",trade.ResultRetcode()," ",trade.ResultRetcodeDescription());
            continue;
        }
        target_changed=true;
        for(int j=0;j<ArraySize(active_trades);j++) if(active_trades[j].ticket==ticket) active_trades[j].take_profit=target;
    }
    if(target_changed) {
        string message=StringFormat("%s %d positions %.2f lots | common TP %s | float %+.2f",direction,count,volume,DoubleToString(target,_Digits),floating);
        AddMLDashboardLog("[BASKET] "+message);
        if(runtime_decision_logs) Print("[NEXA][BASKET] ",message," average=",DoubleToString(average,_Digits));
        WriteDecisionRecord(NewDecisionId(),"BASKET_TP_UPDATED",direction,"SCALP_BASKET",average,0,target,volume,0,0,"",count);
    }
}

string DecisionIdFromComment(string comment,ulong position_id) {
    if(StringFind(comment,"NEXA AI-")==0) {
        string id=StringSubstr(comment,8);
        if(id!="") return id;
    }
    if(StringFind(comment,"NXA-")==0) {
        string id=StringSubstr(comment,4);
        if(id!="") return id;
    }
    return "UNLINKED-"+IntegerToString((int)position_id);
}
int FindTradeByIdentity(ulong order_ticket,ulong deal_ticket,ulong position_id,string comment) {
    for(int i=0;i<ArraySize(active_trades);i++) {
        if(order_ticket>0 && active_trades[i].order_ticket==order_ticket) return i;
        if(deal_ticket>0 && active_trades[i].entry_deal_ticket==deal_ticket) return i;
        if(position_id>0 && active_trades[i].position_id==position_id) return i;
        if(comment!="" && active_trades[i].comment==comment) return i;
    }
    return -1;
}
int FindClosedTradeByIdentity(ulong position_id,ulong position_ticket) {
    for(int i=ArraySize(closed_trades)-1;i>=0;i--)
        if((position_id>0 && closed_trades[i].position_id==position_id) || (position_ticket>0 && closed_trades[i].ticket==position_ticket)) return i;
    return -1;
}
bool IsPositionStillOpen(ulong position_id,ulong position_ticket) {
    if(position_ticket>0 && PositionSelectByTicket(position_ticket)) return true;
    if(position_id==0) return false;
    for(int i=0;i<PositionsTotal();i++) {
        ulong ticket=PositionGetTicket(i);
        if(ticket>0 && (ulong)PositionGetInteger(POSITION_IDENTIFIER)==position_id) return true;
    }
    return false;
}
void RecoverOpenTrades() {
    for(int i=0;i<PositionsTotal();i++) {
        ulong ticket=PositionGetTicket(i);
        if(ticket==0 || (long)PositionGetInteger(POSITION_MAGIC)!=EA_MAGIC_NUMBER || PositionGetString(POSITION_SYMBOL)!=_Symbol) continue;
        ulong position_id=(ulong)PositionGetInteger(POSITION_IDENTIFIER);
        if(FindTradeByIdentity(0,0,position_id,"")>=0) continue;

        TradeInfo recovered;
        recovered.ticket=ticket; recovered.position_id=position_id; recovered.order_ticket=0; recovered.entry_deal_ticket=0;
        recovered.comment=PositionGetString(POSITION_COMMENT);
        recovered.decision_id=DecisionIdFromComment(recovered.comment,position_id);
        recovered.entry_price=PositionGetDouble(POSITION_PRICE_OPEN); recovered.initial_stop_loss=0;
        recovered.stop_loss=PositionGetDouble(POSITION_SL); recovered.take_profit=PositionGetDouble(POSITION_TP);
        recovered.original_lot_size=PositionGetDouble(POSITION_VOLUME); recovered.remaining_lot_size=recovered.original_lot_size;
        recovered.break_even_moved=false; recovered.trailing_active=false; recovered.tp1_hit=false; recovered.tp2_hit=false; recovered.tp3_hit=false;
        recovered.entry_time=(datetime)PositionGetInteger(POSITION_TIME); recovered.trade_type=PositionGetInteger(POSITION_TYPE)==POSITION_TYPE_BUY?0:1;
        recovered.mfe=0; recovered.mae=0; recovered.cumulative_gross=0; recovered.cumulative_commission=0; recovered.cumulative_swap=0;
        recovered.cumulative_net=0; recovered.realized_volume=0; recovered.cumulative_realized_move=0; recovered.partial_info="RECOVERED_AFTER_RESTART"; recovered.last_exit_time=0;
        recovered.final_closing_reason=""; recovered.outcome_written=false; recovered.archived_at=0; recovered.scalp_mode=recovered.stop_loss<=0;
        recovered.ctx_session="UNKNOWN"; recovered.ctx_regime="UNKNOWN"; recovered.ctx_ranging="UNKNOWN"; recovered.ctx_volatility="UNKNOWN";
        recovered.ctx_direction=recovered.trade_type==0?"BUY":"SELL"; recovered.ctx_source="UNKNOWN"; recovered.ctx_mode="UNKNOWN";
        recovered.ctx_snapshot=""; recovered.ctx_key_readings=""; recovered.ctx_bull=0; recovered.ctx_bear=0; recovered.ctx_confirmations=0; recovered.ctx_spread=0;

        if(HistorySelectByPosition(position_id)) {
            for(int d=0;d<HistoryDealsTotal();d++) {
                ulong deal=HistoryDealGetTicket(d);
                if(deal==0 || HistoryDealGetInteger(deal,DEAL_ENTRY)!=DEAL_ENTRY_IN) continue;
                recovered.entry_deal_ticket=deal; recovered.order_ticket=(ulong)HistoryDealGetInteger(deal,DEAL_ORDER);
                recovered.entry_price=HistoryDealGetDouble(deal,DEAL_PRICE); recovered.entry_time=(datetime)HistoryDealGetInteger(deal,DEAL_TIME);
                string deal_comment=HistoryDealGetString(deal,DEAL_COMMENT);
                if(recovered.comment=="" && deal_comment!="") recovered.comment=deal_comment;
                if(StringFind(recovered.decision_id,"UNLINKED-")==0) recovered.decision_id=DecisionIdFromComment(deal_comment,position_id);
                break;
            }
        }
        int size=ArraySize(active_trades); ArrayResize(active_trades,size+1); active_trades[size]=recovered;
    }
}

string JsonEscape(string value) {
    StringReplace(value, "\\", "\\\\");
    StringReplace(value, "\"", "\\\"");
    return value;
}

string TelemetryTimestamp(datetime timestamp) {
    string value = TimeToString(timestamp, TIME_DATE | TIME_SECONDS);
    StringReplace(value, ".", "-");
    StringReplace(value, " ", "T");
    return value + "Z";
}

bool WriteTelemetryHeartbeat(string status) {
    if(IsTesterRun()) return true;   // telemetry file is a live-trading feature
    datetime now = TimeGMT();
    if(status == "ACTIVE" && now == last_telemetry_heartbeat) return true;
    last_telemetry_heartbeat = now;

    string temporary_name = "ea_heartbeat.tmp";
    int file_handle = FileOpen(
        temporary_name,
        FILE_WRITE | FILE_TXT | FILE_ANSI | FILE_COMMON
    );
    if(file_handle == INVALID_HANDLE) {
        if(runtime_debug_logs) {
            Print("Telemetry heartbeat file error: ", GetLastError());
        }
        return false;
    }

    string ea_name = StringFormat("NEXA AI %s",NEXA_VERSION);
    string magic_number = IntegerToString(EA_MAGIC_NUMBER);
    string account_login = IntegerToString(AccountInfoInteger(ACCOUNT_LOGIN));
    string timestamp = TelemetryTimestamp(now);
    string heartbeat = StringFormat(
        "{\"instances\":[{\"name\":\"%s\",\"version\":\"%s\",\"symbol\":\"%s\",\"timeframe\":\"%s\",\"magic_number\":\"%s\",\"status\":\"%s\",\"last_heartbeat\":\"%s\",\"account_login\":\"%s\"}]}",
        JsonEscape(ea_name), EA_VERSION,
        JsonEscape(_Symbol), JsonEscape(EnumToString(_Period)),
        magic_number, status, timestamp, account_login
    );

    FileWriteString(file_handle, heartbeat);
    FileClose(file_handle);

    ResetLastError();
    if(!FileMove(temporary_name, FILE_COMMON, "ea_heartbeat.json", FILE_REWRITE)) {
        if(runtime_debug_logs) {
            Print("Telemetry heartbeat replace error: ", GetLastError());
        }
        return false;
    }

    if(runtime_debug_logs) {
        Print(
            "ANALYSIS | ", _Symbol, " ", EnumToString(_Period),
            " | Status: ", status,
            " | RSI: ", DoubleToString(current_signals.rsi_value, 1),
            " | ATR points: ", DoubleToString(current_signals.atr_value / _Point, 1),
            " | ML trend: ", ml_trend_direction,
            " | Positions: ", PositionsTotal(),
            " | Trades today: ", trades_today, "/", InpMaxTradesPerDay,
            " | UTC: ", timestamp
        );
    }

    return true;
}

bool ExtractMLJsonString(string body,string key,string &value) {
    string needle="\""+key+"\""; int pos=StringFind(body,needle); if(pos<0)return false;
    int colon=StringFind(body,":",pos+StringLen(needle)); if(colon<0)return false;
    int start=StringFind(body,"\"",colon+1); if(start<0)return false;
    int finish=StringFind(body,"\"",start+1); if(finish<0)return false;
    value=StringSubstr(body,start+1,finish-start-1); return true;
}
bool ExtractMLJsonNumber(string body,string key,double &value) {
    string needle="\""+key+"\""; int pos=StringFind(body,needle); if(pos<0)return false;
    int colon=StringFind(body,":",pos+StringLen(needle)); if(colon<0)return false;
    int start=colon+1; while(start<StringLen(body) && (StringGetCharacter(body,start)==' ' || StringGetCharacter(body,start)=='\t'))start++;
    int finish=start; while(finish<StringLen(body)) {
        ushort c=StringGetCharacter(body,finish);
        if(c==',' || c=='}' || c==']' || c==' ' || c=='\r' || c=='\n' || c=='\t')break;
        finish++;
    }
    if(finish<=start)return false; value=StringToDouble(StringSubstr(body,start,finish-start)); return MathIsValidNumber(value);
}

        if(selected_signal=="BUY" || selected_signal=="SELL") {

            double fallback_low=buy_side?ml_potential_bullish_reversal_low:ml_potential_bearish_reversal_low;
            double fallback_high=buy_side?ml_potential_bullish_reversal_high:ml_potential_bearish_reversal_high;
            if(ml_entry_zone_low<=0 || ml_entry_zone_high<ml_entry_zone_low) {
                ml_entry_zone_low=fallback_low;
                ml_entry_zone_high=fallback_high;
                ml_entry_price=(fallback_low>0 && fallback_high>=fallback_low)?(fallback_low+fallback_high)*0.5:0;
                ml_entry_zone_source=buy_side?ml_bullish_reversal_source:ml_bearish_reversal_source;
            }
            bool valid_scalp_zone=ml_entry_zone_low>0 && ml_entry_zone_high>=ml_entry_zone_low;
            bool valid_scalp_candle=ml_entry_candle_low>0 && ml_entry_candle_high>=ml_entry_candle_low;
            ml_entry_triggered=valid_scalp_zone && valid_scalp_candle &&
                ml_entry_candle_low<=ml_entry_zone_high && ml_entry_candle_high>=ml_entry_zone_low;
            ml_planned_direction=selected_signal;
            ml_entry_direction=ml_entry_triggered?selected_signal:"WAIT";
            ml_entry_status=ml_entry_triggered?"ZONE_TOUCHED":valid_scalp_zone?"WAIT_FOR_ZONE":"NO_ENTRY_ZONE";
        } else {
            ml_planned_direction="WAIT";
            ml_entry_direction="WAIT";
            ml_entry_triggered=false;
            ml_entry_status="NO_DIRECTIONAL_EDGE";
        }
    } else if(swing_setup_selected) {
        ml_planned_direction=selected_signal;
    } else {
        selected_signal="WAIT";
    }
    double selected_stop_atr=selected_signal=="BUY"?ml_swing_stop_buy_atr:ml_swing_stop_sell_atr;
    double selected_target_atr=selected_signal=="BUY"?ml_swing_target_buy_atr:ml_swing_target_sell_atr;
    if(selected_mode=="SWING" && (ml_m15_atr<=0 || selected_stop_atr<=0 || selected_target_atr<=selected_stop_atr)) {
        selected_signal="WAIT";
        signal_reason="ML_SWING_EXIT_ESTIMATE_UNAVAILABLE";
    }
    if(selected_mode!="SWING" && selected_mode!="SCALP") selected_mode="WAIT";
    bool scalp_direction_selected=selected_mode=="SCALP" &&
        (selected_signal=="BUY" || selected_signal=="SELL");
    bool scalp_confidence_met=confidence>=scalp_confidence_required;
    bool existing_scalp_grid=selected_mode=="SCALP" && CountEAOpenPositions()>0;
    bool entry_level_ready=ml_entry_triggered || existing_scalp_grid;
    string execution_signal=(selected_signal=="BUY"||selected_signal=="SELL") &&
        ((scalp_direction_selected && scalp_confidence_met && entry_level_ready) ||
         (selected_mode=="SWING" && confidence>=runtime_ml_execution_confidence && ml_entry_triggered))?
        selected_signal:"WAIT";
    if(selected_signal!="BUY" && selected_signal!="SELL") selected_mode="WAIT";
    active_trading_mode=selected_mode;
    ml_prediction_signal=execution_signal; ml_prediction_setup_type=setup_type; ml_prediction_direction=direction;
    ml_prediction_reason=signal_reason;
    ml_prediction_model_version=model_version; ml_prediction_confidence=confidence;
    ml_prediction_time=TimeCurrent();
    ml_prediction_available=true;
    DrawMLPointOfInterest();
    AddMLDashboardLog(StringFormat("MODEL %s %.3f > %s %.3f",
        ml_model_class,ml_model_class_probability,ml_runner_up_class,ml_runner_up_probability));
    AddMLDashboardLog("WHY "+ml_setup_reason);
    AddMLDashboardLog("CONTEXT "+ml_setup_context);

    AddMLDashboardLog(StringFormat("ENTRY %s %s | %s | zone %.5f-%.5f | candle %.5f-%.5f",
        ml_planned_direction,ml_entry_status,ml_entry_triggered?"TOUCHED":"WAIT",
        ml_entry_zone_low,ml_entry_zone_high,ml_entry_candle_low,ml_entry_candle_high));
    if(runtime_decision_logs)
          Print("[NEXA][ML] symbol=",_Symbol," H1 bias=",ml_trend_direction,
              " trend_model=",ml_trend_model_version," setup_model=",ml_h1_model_version,
              " | M15=",setup_type,"/",ml_m15_signal," confidence=",DoubleToString(confidence,1),
              " | confirmation=",ml_confirmation_status," | EA=",execution_signal,
              " reason=",signal_reason," | model=",model_version);
    if(runtime_decision_logs)
        Print("[NEXA][ML][WHY] H1 evidence=",ml_trend_reason," | top class=",ml_model_class,
            " score=",DoubleToString(ml_model_class_probability,3)," runner-up=",ml_runner_up_class,
            " score=",DoubleToString(ml_runner_up_probability,3)," | context=",ml_setup_context,
            " | POI=",ml_poi_reason);
    if(runtime_decision_logs && ml_potential_bullish_reversal_high>0)
        Print("[NEXA][ML] potential bullish reversal reference=",DoubleToString(ml_potential_bullish_reversal_low,_Digits),"-",DoubleToString(ml_potential_bullish_reversal_high,_Digits)," source=",ml_bullish_reversal_source);
    if(runtime_decision_logs && ml_potential_bearish_reversal_high>0)

    if(runtime_decision_logs && (ml_breakout_up>0 || ml_breakout_down>0))
        Print("[NEXA][ML] breakout reference high=",DoubleToString(ml_breakout_up,_Digits)," low=",DoubleToString(ml_breakout_down,_Digits));
    if(runtime_decision_logs)
        Print("[NEXA][ML][ENTRY] status=",ml_entry_status," planned=",ml_planned_direction,
            " entry_direction=",ml_entry_direction,
            " triggered=",ml_entry_triggered," zone=",DoubleToString(ml_entry_zone_low,_Digits),"-",
            DoubleToString(ml_entry_zone_high,_Digits)," level=",DoubleToString(ml_entry_price,_Digits),
            " source=",ml_entry_zone_source," candle=",DoubleToString(ml_entry_candle_low,_Digits),"-",
            DoubleToString(ml_entry_candle_high,_Digits)," execution=",execution_signal);
    Print("[NEXA][ML][OBS] model=",model_version," H1 bias=",ml_trend_direction,
        " M15 setup=",setup_type,"/",ml_m15_signal,

        " planned=",ml_planned_direction,
        " entry=",ml_entry_status," zone=",DoubleToString(ml_entry_zone_low,_Digits),"-",
        DoubleToString(ml_entry_zone_high,_Digits)," candle=",DoubleToString(ml_entry_candle_low,_Digits),"-",
        DoubleToString(ml_entry_candle_high,_Digits)," triggered=",ml_entry_triggered,
        " EA=",execution_signal);
    return true;
}
bool RequestMLPrediction() {
    if(!InpUseMLService || IsTesterRun()) return false;
    string base=MLServiceBaseUrl(); if(StringLen(base)<8) { SetMLLinkState(ML_LINK_NOT_CONFIGURED,0,"ML API base URL not set"); return false; }
    string body=BuildMLCandlePayload(); if(body=="") { SetMLLinkState(ML_LINK_NO_RESPONSE,0,"Fewer than 260 completed H1 or M15 candles available"); return false; }
    char request[]; char response[]; string response_headers=""; string headers=MLHeaders();
    StringToCharArray(body,request,0,StringLen(body),CP_UTF8); ResetLastError(); uint started=GetTickCount();
    SetMLLinkState(ML_LINK_CONNECTING,0,"Requesting current ML analysis");
    int status=WebRequest("POST",base+"/api/ml/predict/consensus",headers,MLRequestTimeoutMs(),request,response,response_headers);
    int error_code=GetLastError(); int elapsed=(int)(GetTickCount()-started);
    if(status==-1) {
        if(error_code==4014) SetMLLinkState(ML_LINK_NOT_PERMITTED,0,"Allow "+base+" in Tools > Options > Expert Advisors");
        else if(error_code==5200) SetMLLinkState(ML_LINK_NOT_CONFIGURED,0,"Endpoint URL is not valid");
        else if(error_code==5202 || elapsed>=MLRequestTimeoutMs())
            SetMLLinkState(ML_LINK_TIMEOUT,0,StringFormat("No answer within %d ms",MLRequestTimeoutMs()));
        else SetMLLinkState(ML_LINK_UNREACHABLE,0,StringFormat("%s (error %d)",base,error_code));
        return false;
    }
    if(status==0) {
        SetMLLinkState(ML_LINK_NO_RESPONSE,0,StringFormat("Empty reply after %d ms",elapsed));
        return false;
    }
    if(status==401) {
        SetMLLinkState(ML_LINK_UNAUTHORIZED,status,"Authentication failed; check InpMLApiKey against Render INGEST_API_KEY");
        return false;
    }
    if(status<200 || status>=300) {
        string detail=status>0?CharArrayToString(response,0,MathMin(160,ArraySize(response))):StringFormat("request error %d",error_code);
        SetMLLinkState(ML_LINK_HTTP_ERROR,status,detail); return false;
    }
    string prediction_body=CharArrayToString(response,0,ArraySize(response));
    if(!AcceptMLPrediction(prediction_body)) { SetMLLinkState(ML_LINK_HTTP_ERROR,status,"Invalid ML prediction response"); return false; }
    ml_link_latency_ms=elapsed; ml_link_last_success=TimeCurrent();     SetMLLinkState(ML_LINK_CONNECTED,status,StringFormat("HTTP %d | %d ms | H1 %s | M15 %s | %s",
        status,elapsed,ml_trend_direction,ml_m15_signal,ml_confirmation_status));
    return true;
}

bool UpdateMLPrediction() {
    if(!runtime_ml_enabled || !InpUseMLService || IsTesterRun()) {
        ClearMLPrediction("ML inference disabled or unavailable");
        ml_consecutive_failures=0;
        ml_retry_not_before=0;
        return false;
    }
    if(ml_inference_in_progress) return false;
    datetime now=TimeCurrent();
    if(now<ml_retry_not_before) return false;
    if(last_ml_inference_attempt>0 && now-last_ml_inference_attempt<MathMax(1,InpMLCheckSeconds)) return false;
    last_ml_inference_attempt=now;
    ml_inference_in_progress=true;
    bool success=RequestMLPrediction();
    ml_inference_in_progress=false;
    if(success) {
        ml_consecutive_failures=0;
        ml_retry_not_before=0;
    } else {
        ClearMLPrediction("Latest ML request failed; waiting for fresh prediction");
        ml_consecutive_failures=(int)MathMin(ml_consecutive_failures+1,5);
        int retry_delay=(int)MathMax(1,InpMLCheckSeconds);
        for(int attempt=1;attempt<ml_consecutive_failures;attempt++)
            retry_delay=(int)MathMin(60,retry_delay*2);
        ml_retry_not_before=now+retry_delay;
    }
    return success;
}

double NormalizeVolume(double volume) {
    double minv=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_MIN),maxv=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_MAX),step=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_STEP);
    if(step<=0)step=minv; volume=MathMax(minv,MathMin(maxv,volume)); volume=MathFloor(volume/step+0.0000001)*step;
    int digits=0; double probe=step; while(digits<8 && MathAbs(probe-MathRound(probe))>0.0000001){probe*=10.0;digits++;} return NormalizeDouble(volume,digits);
}
double CalculateScalpLotSize() {
    int scalp_positions=0;
    for(int i=0;i<PositionsTotal();i++) {
        ulong ticket=PositionGetTicket(i);
        if(ticket==0 || (long)PositionGetInteger(POSITION_MAGIC)!=EA_MAGIC_NUMBER || PositionGetString(POSITION_SYMBOL)!=_Symbol) continue;
        if(PositionGetDouble(POSITION_SL)==0) scalp_positions++;
    }
    double max_lot=0.10;
    double min_lot=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_MIN);
    double volume_step=SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_STEP);
    if(volume_step<=0) volume_step=min_lot;
    double max_allowed=MathFloor(MathMin(SymbolInfoDouble(_Symbol,SYMBOL_VOLUME_MAX),max_lot)/volume_step+0.0000001)*volume_step;
    if(min_lot>max_allowed) {
        Print("[NEXA][SCALP][ERROR] Broker minimum lot ",DoubleToString(min_lot,2)," exceeds the 0.10 lot per-order cap");
        return 0;
    }
    double requested_lot=MathMin(runtime_fixed_lot*MathPow(InpScalpLotMultiplier,scalp_positions),max_allowed);
    double normalized_lot=MathMax(min_lot,MathMin(max_allowed,MathRound(requested_lot/volume_step)*volume_step));
    int volume_digits=0;
    double step_probe=volume_step;
    while(volume_digits<8 && MathAbs(step_probe-MathRound(step_probe))>0.0000001) { step_probe*=10.0; volume_digits++; }
    normalized_lot=NormalizeDouble(normalized_lot,volume_digits);
    if(normalized_lot<=0 || normalized_lot>max_lot) {
        Print("[NEXA][SCALP][ERROR] Unable to size scalp order within the 0.10 lot per-order cap");
        return 0;
    }
    return normalized_lot;
}

}

bool ReadValue(int handle,int buffer,int shift,double &value) { double a[]; ArraySetAsSeries(a,true); if(handle==INVALID_HANDLE || CopyBuffer(handle,buffer,shift,1,a)!=1)return false; value=a[0]; return MathIsValidNumber(value); }
void SetToolState(int idx,double value,double secondary,SignalState state,double strength) { tools[idx].value=value; tools[idx].secondary_value=secondary; tools[idx].state=state; tools[idx].strength=MathMax(0.0,MathMin(1.0,strength)); tools[idx].last_update=iTime(_Symbol,PERIOD_CURRENT,1); }
void UpdateIndicatorSignals() {
    ZeroMemory(current_signals); const int s=1; double v=0,v2=0,v3=0,close1=iClose(_Symbol,PERIOD_CURRENT,s),close2=iClose(_Symbol,PERIOD_CURRENT,s+1);
    if(ReadValue(atr_handle,0,s,v)){ current_signals.atr_value=v; SetToolState(IDX_ATR,v,0,SIGNAL_NEUTRAL,0.5); }
    if(ReadValue(rsi_handle,0,s,v) && ReadValue(rsi_handle,0,s+1,v2)){ current_signals.rsi_value=v; SignalState st=v>v2&&v<InpRSIOverbought?SIGNAL_BULLISH:(v<v2&&v>InpRSIOversold?SIGNAL_BEARISH:SIGNAL_NEUTRAL); SetToolState(IDX_RSI,v,v2,st,MathMin(1.0,MathAbs(v-50.0)/50.0)); }
    if(ReadValue(macd_handle,0,s,v)&&ReadValue(macd_handle,1,s,v2)) SetToolState(IDX_MACD,v,v2,v>v2?SIGNAL_BULLISH:(v<v2?SIGNAL_BEARISH:SIGNAL_NEUTRAL),MathMin(1.0,MathAbs(v-v2)/(current_signals.atr_value+_Point)));
    double adx=0,pdi=0,mdi=0; if(ReadValue(adx_handle,0,s,adx)&&ReadValue(adx_handle,1,s,pdi)&&ReadValue(adx_handle,2,s,mdi)) SetToolState(IDX_ADX,adx,pdi-mdi,pdi>mdi?SIGNAL_BULLISH:(mdi>pdi?SIGNAL_BEARISH:SIGNAL_NEUTRAL),MathMin(1.0,adx/50.0));
    if(ReadValue(stochastic_handle,0,s,v)&&ReadValue(stochastic_handle,1,s,v2)) SetToolState(IDX_STOCHASTIC,v,v2,v>v2&&v<80?SIGNAL_BULLISH:(v<v2&&v>20?SIGNAL_BEARISH:SIGNAL_NEUTRAL),MathMin(1.0,MathAbs(v-v2)/25.0));
    double mid=0,upper=0,lower=0; if(ReadValue(bands_handle,0,s,mid)&&ReadValue(bands_handle,1,s,upper)&&ReadValue(bands_handle,2,s,lower)) SetToolState(IDX_BOLLINGER,close1,mid,close1>mid?SIGNAL_BULLISH:(close1<mid?SIGNAL_BEARISH:SIGNAL_NEUTRAL),(upper>lower)?MathMin(1.0,MathAbs(close1-mid)/(upper-lower)):0);
    if(ReadValue(cci_handle,0,s,v)) SetToolState(IDX_CCI,v,0,v>0?SIGNAL_BULLISH:(v<0?SIGNAL_BEARISH:SIGNAL_NEUTRAL),MathMin(1.0,MathAbs(v)/200.0));
    if(ReadValue(wpr_handle,0,s,v)) SetToolState(IDX_WILLIAMS_R,v,0,v>-50?SIGNAL_BULLISH:(v<-50?SIGNAL_BEARISH:SIGNAL_NEUTRAL),MathMin(1.0,MathAbs(v+50.0)/50.0));
    if(ReadValue(momentum_handle,0,s,v)) SetToolState(IDX_MOMENTUM,v,100,v>100?SIGNAL_BULLISH:(v<100?SIGNAL_BEARISH:SIGNAL_NEUTRAL),MathMin(1.0,MathAbs(v-100.0)/5.0));
    double roc=close2!=0?100.0*(close1-close2)/close2:0; SetToolState(IDX_ROC,roc,0,roc>0?SIGNAL_BULLISH:(roc<0?SIGNAL_BEARISH:SIGNAL_NEUTRAL),MathMin(1.0,MathAbs(roc)/2.0));
    if(ReadValue(obv_handle,0,s,v)&&ReadValue(obv_handle,0,s+1,v2)) SetToolState(IDX_OBV,v,v2,v>v2?SIGNAL_BULLISH:(v<v2?SIGNAL_BEARISH:SIGNAL_NEUTRAL),0.5);
    if(ReadValue(mfi_handle,0,s,v)) SetToolState(IDX_MFI,v,0,v>50&&v<80?SIGNAL_BULLISH:(v<50&&v>20?SIGNAL_BEARISH:SIGNAL_NEUTRAL),MathMin(1.0,MathAbs(v-50.0)/50.0));
    double tenkan=0,kijun=0,spanA=0,spanB=0; if(ReadValue(ichimoku_handle,0,s,tenkan)&&ReadValue(ichimoku_handle,1,s,kijun)&&ReadValue(ichimoku_handle,2,s,spanA)&&ReadValue(ichimoku_handle,3,s,spanB)) SetToolState(IDX_ICHIMOKU,tenkan,kijun,(tenkan>kijun&&close1>MathMax(spanA,spanB))?SIGNAL_BULLISH:((tenkan<kijun&&close1<MathMin(spanA,spanB))?SIGNAL_BEARISH:SIGNAL_NEUTRAL),0.7);
    if(ReadValue(sar_handle,0,s,v)) SetToolState(IDX_SAR,v,close1,close1>v?SIGNAL_BULLISH:(close1<v?SIGNAL_BEARISH:SIGNAL_NEUTRAL),current_signals.atr_value>0?MathMin(1.0,MathAbs(close1-v)/current_signals.atr_value):0);
    UpdateCalculatedTools(); CheckPriceActionSignals();
    if(runtime_indicator_logs && TimeCurrent()-last_indicator_print>=60){ last_indicator_print=TimeCurrent(); Print("[NEXA][INDICATORS] ",KeyReadings()); }
}
void UpdateCalculatedTools() {
    const int count=20; MqlRates rates[]; ArraySetAsSeries(rates,true); if(CopyRates(_Symbol,PERIOD_CURRENT,1,count,rates)<count)return;
    double pv=0,vol=0,highest=rates[1].high,lowest=rates[1].low;
    for(int i=1;i<count;i++){ double typical=(rates[i].high+rates[i].low+rates[i].close)/3.0; pv+=typical*(double)rates[i].tick_volume; vol+=(double)rates[i].tick_volume; highest=MathMax(highest,rates[i].high); lowest=MathMin(lowest,rates[i].low); }
    double vwap=vol>0?pv/vol:rates[0].close; SetToolState(IDX_VWAP,vwap,rates[0].close,rates[0].close>vwap?SIGNAL_BULLISH:(rates[0].close<vwap?SIGNAL_BEARISH:SIGNAL_NEUTRAL),0.5);
    double middle=(highest+lowest)/2.0; SetToolState(IDX_DONCHIAN,rates[0].close,middle,rates[0].close>middle?SIGNAL_BULLISH:(rates[0].close<middle?SIGNAL_BEARISH:SIGNAL_NEUTRAL),(highest>lowest)?MathMin(1.0,MathAbs(rates[0].close-middle)/(highest-lowest)):0);
    double atr=tools[IDX_ATR].value, basis=(rates[1].high+rates[1].low)/2.0; double upper=basis+3.0*atr,lower=basis-3.0*atr;
    SignalState st=rates[0].close>upper?SIGNAL_BULLISH:(rates[0].close<lower?SIGNAL_BEARISH:SIGNAL_NEUTRAL); SetToolState(IDX_SUPERTREND,st==SIGNAL_BULLISH?lower:upper,basis,st,0.65);
}

//+------------------------------------------------------------------+
//| Check price action signals                                    |
//+------------------------------------------------------------------+
void CheckPriceActionSignals() {
    MqlRates r[]; ArraySetAsSeries(r,true); if(CopyRates(_Symbol,PERIOD_CURRENT,1,3,r)<3)return;
    bool bull=(r[1].close<r[1].open&&r[0].close>r[0].open&&r[0].open<r[1].close&&r[0].close>r[1].open);
    bool bear=(r[1].close>r[1].open&&r[0].close<r[0].open&&r[0].open>r[1].close&&r[0].close<r[1].open);
    double body=MathMax(_Point,MathAbs(r[0].close-r[0].open)),lower=MathMin(r[0].close,r[0].open)-r[0].low,upper=r[0].high-MathMax(r[0].close,r[0].open);
    bull=bull||(lower>body*2.0&&upper<body*0.5); bear=bear||(upper>body*2.0&&lower<body*0.5);
    SetToolState(IDX_PRICE_ACTION,r[0].close,body,bull?SIGNAL_BULLISH:(bear?SIGNAL_BEARISH:SIGNAL_NEUTRAL),bull||bear?0.8:0);
}
//+------------------------------------------------------------------+
//| Check if trading is allowed                                   |
//+------------------------------------------------------------------+
bool IsTradeAllowed() {
    if(!runtime_trading_enabled) return false;

    // Check terminal and account permissions
    if(!TerminalInfoInteger(TERMINAL_TRADE_ALLOWED) || !MQLInfoInteger(MQL_TRADE_ALLOWED)) {
        return false;
    }
    
    // Check daily trade limit
    if(trades_today >= InpMaxTradesPerDay) {
        if(runtime_decision_logs) Print("[BLOCK] Daily trade limit reached: ", trades_today);
        return false;
    }
    
    // Check hourly trade limit
    if(trades_this_hour >= InpMaxTradesPerHour) {
        if(runtime_decision_logs) Print("[BLOCK] Hourly trade limit reached: ", trades_this_hour);
        return false;
    }
    
    // Check session filter
    if(runtime_use_session && !IsWithinTradingSession()) {
        return false;
    }
    
    // Check spread filter
    if(InpUseSpreadFilter && !IsSpreadAcceptable()) {
        return false;
    }
    
    // Check time between trades
    if(runtime_use_news && IsNewsBlocked()) { if(runtime_decision_logs) Print("[BLOCK] High-impact news window active"); return false; }
    if(TimeCurrent() - last_trade_time < InpMinTimeBetweenTrades * 60) {
        if(runtime_decision_logs) Print("[BLOCK] Too soon since last trade");
        return false;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| Check if within trading session                               |
//+------------------------------------------------------------------+
bool IsWithinTradingSession() {
    MqlDateTime dt; TimeToStruct(TimeCurrent(),dt);
    bool generic=HourInWindow(dt.hour,runtime_session_start,runtime_session_end);
    bool named=IsCurrentSessionEnabled();
    if(!generic || !named){ if(runtime_decision_logs) Print("Outside configured session: ",GetCurrentSession()," broker hour ",dt.hour); return false; }
    return true;
}

//+------------------------------------------------------------------+
//| Check if spread is acceptable                                 |
//+------------------------------------------------------------------+
bool IsSpreadAcceptable() {
    double spread = (SymbolInfoDouble(_Symbol, SYMBOL_ASK) - SymbolInfoDouble(_Symbol, SYMBOL_BID)) / _Point;
    
    if(spread > InpMaxSpreadPips) {
        if(runtime_decision_logs) {
            Print("[BLOCK] Spread too high: ", DoubleToString(spread, 1), " pips");
        }
        return false;
    }
    
    return true;
}

//+------------------------------------------------------------------+
//| Should look for new trades                                    |
//+------------------------------------------------------------------+
bool ShouldLookForNewTrades() {
    if(!runtime_new_trades_enabled) return false;
    if(!runtime_ml_enabled) return false;
    if(runtime_ml_enabled && active_trading_mode!="SWING" && active_trading_mode!="SCALP") return false;
    int position_limit=IsScalpMode()?InpScalpMaxPositions:InpMaxOpenPositions;
    if(CountEAOpenPositions()>=position_limit){ if(runtime_decision_logs)Print("EA position limit reached: ",position_limit); return false; }
    if(CountEAOpenPositions()==0 && !MLCurrentPriceInsideEntryZone()) {
        if(runtime_decision_logs && ml_entry_triggered)
            Print("[NEXA][ML][ENTRY] candle touched zone but executable quote is outside it; waiting for a live zone touch");
        return false;
    }
    if(IsScalpMode()) {
        if(!IsGoldSymbol()) return false;
        if(!scalp_basket_tp_ready) return false;
        if(!GridAddIsDue(ml_prediction_signal)) return false;
    }
    return true;
}
bool GridAddIsDue(string direction) {
    if(direction!="BUY" && direction!="SELL") return false;
    bool found=false; double anchor=0;
    for(int i=0;i<PositionsTotal();i++) {
        ulong ticket=PositionGetTicket(i);
        if(ticket==0 || (long)PositionGetInteger(POSITION_MAGIC)!=EA_MAGIC_NUMBER || PositionGetString(POSITION_SYMBOL)!=_Symbol) continue;
        if(PositionGetDouble(POSITION_SL)>0) continue;
        long type=PositionGetInteger(POSITION_TYPE);
        if((direction=="BUY" && type!=POSITION_TYPE_BUY) || (direction=="SELL" && type!=POSITION_TYPE_SELL)) return false;
        double entry=PositionGetDouble(POSITION_PRICE_OPEN);
        if(!found) anchor=entry;
        else anchor=direction=="BUY"?MathMin(anchor,entry):MathMax(anchor,entry);
        found=true;
    }
    if(!found) return true;
    double spacing=ScalpGridSpacing(direction);
    double market_price=direction=="BUY"?SymbolInfoDouble(_Symbol,SYMBOL_BID):SymbolInfoDouble(_Symbol,SYMBOL_ASK);
    return direction=="BUY"?market_price<=anchor-spacing:market_price>=anchor+spacing;
}
bool MLCurrentPriceInsideEntryZone() {
    if(!ml_entry_triggered || ml_entry_zone_low<=0 || ml_entry_zone_high<ml_entry_zone_low) return false;
    if(ml_planned_direction!="BUY" && ml_planned_direction!="SELL") return false;
    double executable_price=ml_planned_direction=="BUY"?SymbolInfoDouble(_Symbol,SYMBOL_ASK):
        SymbolInfoDouble(_Symbol,SYMBOL_BID);
    return executable_price>=ml_entry_zone_low && executable_price<=ml_entry_zone_high;
}
string ModeReadinessText() {
    if(!runtime_trading_enabled) return "PENDING  |  trading is stopped";
    if(!runtime_new_trades_enabled) return "PENDING  |  new trades are disabled";
    if(!runtime_ml_enabled || !InpUseMLService) return "PENDING  |  NEXA ML is unavailable";
    if(!ml_prediction_available) return "PENDING  |  waiting for a fresh ML prediction";
    if(active_trading_mode!="SWING" && active_trading_mode!="SCALP")
        return "PENDING  |  "+(ml_prediction_reason!=""?ml_prediction_reason:"no valid ML setup");
    if(active_trading_mode=="SCALP" &&
       ml_prediction_confidence<MathMax(SCALP_MIN_ML_CONFIDENCE,runtime_ml_execution_confidence))
        return StringFormat("PENDING  |  scalp confidence %.1f%% below %.0f%% minimum",
            ml_prediction_confidence,MathMax(SCALP_MIN_ML_CONFIDENCE,runtime_ml_execution_confidence));
    if(active_trading_mode=="SWING" && ml_prediction_confidence<runtime_ml_execution_confidence)
        return StringFormat("PENDING  |  setup confidence %.1f%% below %.0f%% minimum",
            ml_prediction_confidence,runtime_ml_execution_confidence);
    if(CountEAOpenPositions()==0 && !MLCurrentPriceInsideEntryZone()) {
        string entry_wait=ml_entry_triggered?"candle touched; live quote outside zone":ml_entry_status;
        return "PENDING  |  ML "+entry_wait+" "+
            DoubleToString(ml_entry_zone_low,_Digits)+"-"+DoubleToString(ml_entry_zone_high,_Digits);
    }
    if((ml_prediction_signal=="BUY" && !runtime_buy_enabled) || (ml_prediction_signal=="SELL" && !runtime_sell_enabled))
        return "PENDING  |  selected direction is disabled";
    if(!TerminalInfoInteger(TERMINAL_TRADE_ALLOWED) || !MQLInfoInteger(MQL_TRADE_ALLOWED)) return "PENDING  |  terminal/account trading permission";
    if(trades_today>=InpMaxTradesPerDay) return "PENDING  |  daily trade limit reached";
    if(trades_this_hour>=InpMaxTradesPerHour) return "PENDING  |  hourly trade limit reached";
    if(TimeCurrent()-last_trade_time<InpMinTimeBetweenTrades*60) return "PENDING  |  minimum trade interval";
    if(InpMinATRPoints>0 && current_signals.atr_value/_Point<InpMinATRPoints) return "PENDING  |  ATR minimum gate";
    if(runtime_use_session) {
        MqlDateTime session_time; TimeToStruct(TimeCurrent(),session_time);
        if(!HourInWindow(session_time.hour,runtime_session_start,runtime_session_end) || !IsCurrentSessionEnabled())
            return "PENDING  |  outside enabled session";
    }
    double current_spread=(SymbolInfoDouble(_Symbol,SYMBOL_ASK)-SymbolInfoDouble(_Symbol,SYMBOL_BID))/_Point;
    if(InpUseSpreadFilter && current_spread>InpMaxSpreadPips) return "PENDING  |  spread filter";
    if(runtime_use_news && IsNewsBlocked()) return "PENDING  |  high-impact news window";
    if(active_trading_mode=="SWING") {
        if(CountEAOpenPositions()>=InpMaxOpenPositions) return "PENDING  |  swing position limit reached";
        return "READY  |  H1/M15 setup and ML exits accepted";
    }
    if(!IsGoldSymbol()) return "PENDING  |  scalp is restricted to XAUUSD";
    if(!scalp_basket_tp_ready) return "PENDING  |  synchronizing common basket TP";
    if(CountEAOpenPositions()>=InpScalpMaxPositions) return "PENDING  |  scalp position limit reached";
    if(!GridAddIsDue(ml_prediction_signal))
        return StringFormat("PENDING  |  waiting for %s ML add interval %.2f",
            ml_prediction_signal,ScalpGridSpacing(ml_prediction_signal));
    return "READY  |  XAUUSD ML entry/spacing and basket TP accepted";
}

void ScanForTradingOpportunities() {
    if(!ml_prediction_available || (ml_prediction_signal!="BUY" && ml_prediction_signal!="SELL")) {

            Print("[NEXA][ML] Entry rejected: no confirmed directional ML prediction");

    pending_decision_id=NewDecisionId();
    StageDecisionContext(ml_prediction_signal,"ML_ONLY",0,0,0);
    ExecuteHighProbabilityTrade(order_type,0,"ML_ONLY");
}

    stage_session=GetCurrentSession(); stage_regime=GetMarketRegime(); stage_ranging=RangingText();
    stage_volatility=GetVolatilityRegime(); stage_direction=direction; stage_source=source; stage_mode=OperatingModeText()+"_"+active_trading_mode+"_"+MLModeText();

//+------------------------------------------------------------------+
//| Execute high probability trade                                |
//+------------------------------------------------------------------+
void ExecuteHighProbabilityTrade(ENUM_ORDER_TYPE trade_type,int signal_strength,string trade_source) {
    if((trade_type==ORDER_TYPE_BUY&&!runtime_buy_enabled)||(trade_type==ORDER_TYPE_SELL&&!runtime_sell_enabled)||!runtime_new_trades_enabled)return;
    if(runtime_ml_enabled) {
        string requested_side=trade_type==ORDER_TYPE_BUY?"BUY":"SELL";
        if(!ml_prediction_available || ml_prediction_signal!=requested_side ||
           ml_prediction_direction!=requested_side || ml_planned_direction!=requested_side) {
            Print("[NEXA][ERROR] Blocked ML direction mismatch | requested=",requested_side,
                " signal=",ml_prediction_signal," direction=",ml_prediction_direction,
                " planned=",ml_planned_direction," source=",trade_source);
            return;
        }
    }
    double entry_price=trade_type==ORDER_TYPE_BUY?SymbolInfoDouble(_Symbol,SYMBOL_ASK):SymbolInfoDouble(_Symbol,SYMBOL_BID),stop_loss=0,take_profit=0;
    CalculateStopLossAndTakeProfit(trade_type,entry_price,stop_loss,take_profit);
    double lot_size=IsScalpMode()?CalculateScalpLotSize():CalculateLotSize(entry_price,stop_loss);
    if(lot_size<=0||(!IsScalpMode()&&stop_loss<=0)||take_profit<=0){ Print("Invalid order parameters for ",active_trading_mode," mode"); return; }
    if(pending_decision_id=="") pending_decision_id=NewDecisionId();
    string comment="NEXA AI-"+pending_decision_id;
    if(StringLen(comment)>31) {
        Print("[NEXA][ERROR] Generated trade comment exceeds the 31-character broker limit.");
        return;
    }
    if(stage_direction=="" || stage_direction!=(trade_type==ORDER_TYPE_BUY?"BUY":"SELL"))
        StageDecisionContext(trade_type==ORDER_TYPE_BUY?"BUY":"SELL",trade_source,stage_bull,stage_bear,signal_strength);
    WriteDecisionRecord(pending_decision_id,"ENTRY_REQUEST",trade_type==ORDER_TYPE_BUY?"BUY":"SELL",trade_source,entry_price,stop_loss,take_profit,lot_size,stage_bull,stage_bear,"",signal_strength);
    bool result=trade_type==ORDER_TYPE_BUY?trade.Buy(lot_size,_Symbol,0,stop_loss,take_profit,comment):trade.Sell(lot_size,_Symbol,0,stop_loss,take_profit,comment);
        if(result) {
          double fill_price=trade.ResultPrice();
          if(fill_price<=0) fill_price=entry_price;
          AddTradeToMonitoring(trade.ResultOrder(),entry_price,stop_loss,take_profit,lot_size,trade_type,comment);
          RecoverOpenTrades(); UpdateScalpBasketTakeProfit();
          if(IsScalpMode()) {
              int basket_count=0; double basket_volume=0,basket_average=0,basket_target=0,basket_floating=0; string basket_direction="";
              if(GetScalpBasketStats(basket_count,basket_volume,basket_average,basket_target,basket_floating,basket_direction)) { stop_loss=0; take_profit=basket_target; }
          }
          trades_today++; trades_this_hour++; last_trade_time=TimeCurrent();
          string executed_side=trade_type==ORDER_TYPE_BUY?"BUY":"SELL";
          Print("[NEXA][TRADE] EXECUTED | id=",pending_decision_id," symbol=",_Symbol," side=",executed_side,
              " source=",trade_source," order=",trade.ResultOrder()," lots=",DoubleToString(lot_size,2),
              " fill=",DoubleToString(fill_price,_Digits)," SL=",DoubleToString(stop_loss,_Digits)," TP=",DoubleToString(take_profit,_Digits));
          AddMLDashboardLog(StringFormat("TRADE FILLED %s @ %s | SL %s TP %s",executed_side,DoubleToString(fill_price,_Digits),DoubleToString(stop_loss,_Digits),DoubleToString(take_profit,_Digits)));
          if(runtime_ml_enabled && ml_prediction_available) {
            Print("[NEXA][TRADE][ML] H1 bias=",ml_trend_direction,
                " M15 setup=",ml_prediction_setup_type,"/",ml_m15_signal,
                " confirmation=",ml_confirmation_status," entry=",ml_entry_status,
                " model=",ml_prediction_model_version);
            if(ml_potential_bullish_reversal_high>0 || ml_potential_bearish_reversal_high>0)
                Print("[NEXA][TRADE][ML] reversal_refs bull=",DoubleToString(ml_potential_bullish_reversal_low,_Digits),"-",DoubleToString(ml_potential_bullish_reversal_high,_Digits),
                    " (",ml_bullish_reversal_source,") bear=",DoubleToString(ml_potential_bearish_reversal_low,_Digits),"-",DoubleToString(ml_potential_bearish_reversal_high,_Digits),
                    " (",ml_bearish_reversal_source,")");
            Print("[NEXA][TRADE][ML] breakout_refs up=",DoubleToString(ml_breakout_up,_Digits)," down=",DoubleToString(ml_breakout_down,_Digits));
          }
        }
    else {
        Print("Trade execution failed: ",trade.ResultRetcode()," ",trade.ResultRetcodeDescription());
        if(InpLogRejectedDecisions) WriteDecisionRecord(pending_decision_id,"ENTRY_FAILED",trade_type==ORDER_TYPE_BUY?"BUY":"SELL",trade_source,entry_price,stop_loss,take_profit,lot_size,signal_strength,0,"BROKER_RETCODE_"+IntegerToString(trade.ResultRetcode()),signal_strength);
    }
    pending_decision_id=""; stage_direction="";
}

//+------------------------------------------------------------------+
//| Calculate stop loss and take profit                           |
//+------------------------------------------------------------------+
void CalculateStopLossAndTakeProfit(ENUM_ORDER_TYPE trade_type, double entry_price, double &stop_loss, double &take_profit) {
    if(IsScalpMode()) {
        stop_loss=0;
        take_profit=NormalizeDouble(entry_price+(trade_type==ORDER_TYPE_BUY?1.0:-1.0)*InpScalpTakeProfitPips*ScalpPipSize(),_Digits);
        return;
    }
    if(runtime_ml_enabled && active_trading_mode=="SWING") {
        double stop_atr=trade_type==ORDER_TYPE_BUY?ml_swing_stop_buy_atr:ml_swing_stop_sell_atr;
        double target_atr=trade_type==ORDER_TYPE_BUY?ml_swing_target_buy_atr:ml_swing_target_sell_atr;
        stop_loss=NormalizeDouble(entry_price+(trade_type==ORDER_TYPE_BUY?-1.0:1.0)*stop_atr*ml_m15_atr,_Digits);
        take_profit=NormalizeDouble(entry_price+(trade_type==ORDER_TYPE_BUY?1.0:-1.0)*target_atr*ml_m15_atr,_Digits);
        return;
    }
    if(InpUseATRStops) {
        // ATR-based SL/TP
        double atr_values[];
        if(CopyBuffer(atr_handle, 0, 1, 1, atr_values) >= 1) {
            double atr = atr_values[0];
            
            if(trade_type == ORDER_TYPE_BUY) {
                stop_loss = entry_price - (atr * InpATRMultiplierSL);
                take_profit = entry_price + (atr * InpATRMultiplierTP);
            } else {
                stop_loss = entry_price + (atr * InpATRMultiplierSL);
                take_profit = entry_price - (atr * InpATRMultiplierTP);
            }
        }
    } else {
        // Pip-based SL/TP
        double pip_value = (_Digits==3 || _Digits==5) ? _Point*10.0 : _Point;
        
        if(trade_type == ORDER_TYPE_BUY) {
            stop_loss = entry_price - (InpStopLossPips * pip_value);
            take_profit = entry_price + (InpTakeProfitPips * pip_value);
        } else {
            stop_loss = entry_price + (InpStopLossPips * pip_value);
            take_profit = entry_price - (InpTakeProfitPips * pip_value);
        }
    }
    
    // Normalize prices
    stop_loss = NormalizeDouble(stop_loss, _Digits);
    take_profit = NormalizeDouble(take_profit, _Digits);
}

//+------------------------------------------------------------------+
//| Calculate lot size based on risk                              |
//+------------------------------------------------------------------+
double CalculateLotSize(double entry_price, double stop_loss_price) {
    if(IsScalpMode()) return NormalizeVolume(runtime_fixed_lot);
    if(!runtime_use_risk_lots) return NormalizeVolume(runtime_fixed_lot);
    double risk_amount=AccountInfoDouble(ACCOUNT_BALANCE)*runtime_risk_percentage/100.0,loss_for_one_lot=0;
    ENUM_ORDER_TYPE side=stop_loss_price<entry_price?ORDER_TYPE_BUY:ORDER_TYPE_SELL;
    if(!OrderCalcProfit(side,_Symbol,1.0,entry_price,stop_loss_price,loss_for_one_lot)){ Print("[NEXA][ERROR] Lot calculation failed: ",GetLastError());return 0; }
    loss_for_one_lot=MathAbs(loss_for_one_lot); if(loss_for_one_lot<=0)return 0; double lots=NormalizeVolume(risk_amount/loss_for_one_lot);
    if(runtime_decision_logs)Print("[NEXA][RISK] Balance=",DoubleToString(AccountInfoDouble(ACCOUNT_BALANCE),2)," Risk=",DoubleToString(runtime_risk_percentage,2),"% Lots=",DoubleToString(lots,2)); return lots;
}

//+------------------------------------------------------------------+
//| Manage existing trades                                        |
//+------------------------------------------------------------------+
void ManageExistingTrades() {
    for(int i = ArraySize(active_trades) - 1; i >= 0; i--) {
        if(!position.SelectByTicket(active_trades[i].ticket)) {
            // Position no longer open: archive so the exit record stays linkable to the entry
            ArchiveTrade(i);
            continue;
        }
        
        double tracked_price = (active_trades[i].trade_type == 0) ? SymbolInfoDouble(_Symbol, SYMBOL_BID) : SymbolInfoDouble(_Symbol, SYMBOL_ASK);
        double excursion = (active_trades[i].trade_type == 0) ? tracked_price - active_trades[i].entry_price : active_trades[i].entry_price - tracked_price;
        active_trades[i].mfe = MathMax(active_trades[i].mfe, excursion);
        active_trades[i].mae = MathMin(active_trades[i].mae, excursion);

        if(active_trades[i].scalp_mode) continue;

        // Handle partial take profits
        if(runtime_use_partial_tp) {
            HandlePartialTakeProfits(i);
        }
        
        // Handle breakeven
        if(runtime_use_break_even && !active_trades[i].break_even_moved) {
            HandleBreakEven(i);
        }
        
        // Handle trailing stop
        if(runtime_use_trailing && active_trades[i].break_even_moved) {
            HandleTrailingStop(i);
        }
    }
}

//+------------------------------------------------------------------+
//| Handle partial take profits                                   |
//+------------------------------------------------------------------+
void HandlePartialTakeProfits(int trade_index) {
    if(!position.SelectByTicket(active_trades[trade_index].ticket)) return;
    
    double current_price = (active_trades[trade_index].trade_type == 0) ? 
                          SymbolInfoDouble(_Symbol, SYMBOL_BID) : 
                          SymbolInfoDouble(_Symbol, SYMBOL_ASK);
    
    double entry_price = active_trades[trade_index].entry_price;
    double stop_loss = active_trades[trade_index].stop_loss;
    double risk_distance = MathAbs(entry_price - stop_loss);
    
    // Calculate profit in RR terms
    double profit_distance = (active_trades[trade_index].trade_type == 0) ? 
                            (current_price - entry_price) : 
                            (entry_price - current_price);
    
    double current_rr = risk_distance>0 ? profit_distance/risk_distance : 0;
    
    // TP1
    if(!active_trades[trade_index].tp1_hit && current_rr >= InpTP1RR) {
        double close_volume = active_trades[trade_index].remaining_lot_size * InpTP1Percentage / 100.0;
        if(ExecutePartialClose(active_trades[trade_index].ticket, close_volume, "TP1")) {
            active_trades[trade_index].tp1_hit = true;
            active_trades[trade_index].remaining_lot_size -= close_volume;
            Print("[OK] TP1 Hit - Closed ", InpTP1Percentage, "% at RR:", DoubleToString(current_rr, 2));
        }
    }
    
    // TP2
    if(active_trades[trade_index].tp1_hit && !active_trades[trade_index].tp2_hit && current_rr >= InpTP2RR) {
        double close_volume = active_trades[trade_index].original_lot_size * InpTP2Percentage / 100.0;
        if(ExecutePartialClose(active_trades[trade_index].ticket, close_volume, "TP2")) {
            active_trades[trade_index].tp2_hit = true;
            active_trades[trade_index].remaining_lot_size -= close_volume;
            Print("[OK] TP2 Hit - Closed ", InpTP2Percentage, "% at RR:", DoubleToString(current_rr, 2));
        }
    }
    
    // TP3 (close remaining position)
    if(active_trades[trade_index].tp2_hit && !active_trades[trade_index].tp3_hit && current_rr >= InpTP3RR) {
        if(ExecutePartialClose(active_trades[trade_index].ticket, active_trades[trade_index].remaining_lot_size, "TP3")) {
            active_trades[trade_index].tp3_hit = true;
            active_trades[trade_index].remaining_lot_size = 0;
            Print("[OK] TP3 Hit - Closed remaining position at RR:", DoubleToString(current_rr, 2));
        }
    }
}

//+------------------------------------------------------------------+
//| Execute partial close                                         |
//+------------------------------------------------------------------+
bool ExecutePartialClose(ulong ticket, double volume, string comment) {
    if(volume <= 0) return false;
    if(!position.SelectByTicket(ticket)) return false;

    double open_volume = position.Volume();
    double min_lot  = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_MIN);
    double lot_step = SymbolInfoDouble(_Symbol, SYMBOL_VOLUME_STEP);
    if(lot_step <= 0) lot_step = min_lot;

    volume = MathMin(volume, open_volume);
    volume = MathFloor(volume / lot_step) * lot_step;
    volume = NormalizeDouble(volume, 2);

    if(volume < min_lot) return false;                       // nothing valid to close
    if(open_volume - volume < min_lot) volume = open_volume; // avoid an unclosable remainder

    bool ok = trade.PositionClosePartial(ticket, volume);
    if(!ok) Print("Partial close rejected (", comment, "): ", trade.ResultRetcode(), " ", trade.ResultRetcodeDescription());
    return ok;
}

//+------------------------------------------------------------------+
//| Handle breakeven                                              |
//+------------------------------------------------------------------+
void HandleBreakEven(int trade_index) {
    if(!position.SelectByTicket(active_trades[trade_index].ticket)) return;
    
    double current_price = (active_trades[trade_index].trade_type == 0) ? 
                          SymbolInfoDouble(_Symbol, SYMBOL_BID) : 
                          SymbolInfoDouble(_Symbol, SYMBOL_ASK);
    
    double entry_price = active_trades[trade_index].entry_price;
    double stop_loss = active_trades[trade_index].stop_loss;
    
    // Check if profit threshold is reached
    bool should_move_be = false;
    
    if(InpUseATRStops) {
        // RR-based breakeven
        double risk_distance = MathAbs(entry_price - stop_loss);
        double profit_distance = (active_trades[trade_index].trade_type == 0) ? 
                                (current_price - entry_price) : 
                                (entry_price - current_price);
        
        double current_rr = risk_distance>0 ? profit_distance/risk_distance : 0;
        should_move_be = (current_rr >= InpBreakEvenTriggerRR);
    } else {
        // Pip-based breakeven
        double profit_pips = MathAbs(current_price - entry_price) / _Point;
        should_move_be = (profit_pips >= InpBreakEvenTriggerPips);
    }
    
    if(should_move_be) {
        double new_sl = active_trades[trade_index].trade_type==0 ? entry_price+(2*_Point) : entry_price-(2*_Point); // buffer in the profitable direction
        new_sl = NormalizeDouble(new_sl, _Digits);
        
        if(trade.PositionModify(active_trades[trade_index].ticket, new_sl, position.TakeProfit())) {
            active_trades[trade_index].break_even_moved = true;
            active_trades[trade_index].stop_loss = new_sl;
            Print("[OK] Moved to breakeven - Ticket: ", active_trades[trade_index].ticket);
        }
    }
}

//+------------------------------------------------------------------+
//| Handle trailing stop                                          |
//+------------------------------------------------------------------+
void HandleTrailingStop(int trade_index) {
    if(!position.SelectByTicket(active_trades[trade_index].ticket)) return;
    
    double current_price = (active_trades[trade_index].trade_type == 0) ? 
                          SymbolInfoDouble(_Symbol, SYMBOL_BID) : 
                          SymbolInfoDouble(_Symbol, SYMBOL_ASK);
    
    double entry_price = active_trades[trade_index].entry_price;
    double current_sl = position.StopLoss();
    
    // Calculate new trailing stop
    double new_sl = current_sl;
    bool should_trail = false;
    
    if(InpUseATRStops) {
        // RR-based trailing
        double risk_distance = MathAbs(entry_price - active_trades[trade_index].stop_loss);
        double trail_distance = risk_distance * InpTrailingStepRR;
        
        if(active_trades[trade_index].trade_type == 0) { // Buy
            new_sl = current_price - trail_distance;
            should_trail = (new_sl > current_sl);
        } else { // Sell
            new_sl = current_price + trail_distance;
            should_trail = (new_sl < current_sl);
        }
    } else {
        // Pip-based trailing
        double trail_distance = InpTrailingStepPips * _Point;
        
        if(active_trades[trade_index].trade_type == 0) { // Buy
            new_sl = current_price - trail_distance;
            should_trail = (new_sl > current_sl);
        } else { // Sell
            new_sl = current_price + trail_distance;
            should_trail = (new_sl < current_sl);
        }
    }
    
    if(should_trail) {
        new_sl = NormalizeDouble(new_sl, _Digits);
        
        if(trade.PositionModify(active_trades[trade_index].ticket, new_sl, position.TakeProfit())) {
            active_trades[trade_index].stop_loss = new_sl;
            Print("[TRAIL] Trailing stop updated - Ticket: ", active_trades[trade_index].ticket, " New SL: ", new_sl);
        }
    }
}

//+------------------------------------------------------------------+

//+------------------------------------------------------------------+
void RemoveTradeFromArray(int index) {
    if(index < 0 || index >= ArraySize(active_trades)) return;

        active_trades[i] = active_trades[i + 1];
    }
    ArrayResize(active_trades, ArraySize(active_trades) - 1);
}

//| Update counters                                               |
//+------------------------------------------------------------------+
void UpdateCounters() {
    datetime current_time = TimeCurrent();
    MqlDateTime dt;
    TimeToStruct(current_time, dt);
    
    // Reset daily counter
    datetime current_date = current_time - (current_time % 86400);
    if(current_date != last_trade_date) {
        trades_today = 0;
        last_trade_date = current_date;
        if(runtime_debug_logs) Print("[DAY] New trading day - Counters reset");
    }
    
    // Reset hourly counter
    datetime current_hour = current_time - (current_time % 3600);
    if(current_hour != last_hour_reset) {
        trades_this_hour = 0;
        last_hour_reset = current_hour;
        if(runtime_debug_logs) Print("[HOUR] New trading hour - Hour counter reset");
    }
}

    last_trade_date = TimeCurrent() - (TimeCurrent() % 86400);

    last_trade_time = 0;
}

void OnTradeTransaction(const MqlTradeTransaction &trans,const MqlTradeRequest &request,const MqlTradeResult &result) {
    if(trans.type!=TRADE_TRANSACTION_DEAL_ADD || trans.deal==0)return;
    if(!HistoryDealSelect(trans.deal))return;
    if((long)HistoryDealGetInteger(trans.deal,DEAL_MAGIC)!=EA_MAGIC_NUMBER)return;
    long entry_type=HistoryDealGetInteger(trans.deal,DEAL_ENTRY); ulong position_id=(ulong)HistoryDealGetInteger(trans.deal,DEAL_POSITION_ID);
    ulong order_ticket=(ulong)HistoryDealGetInteger(trans.deal,DEAL_ORDER); string deal_comment=HistoryDealGetString(trans.deal,DEAL_COMMENT);
    if(entry_type==DEAL_ENTRY_IN){
        int entry_index=FindTradeByIdentity(order_ticket,trans.deal,position_id,deal_comment);
        if(entry_index>=0) {
            active_trades[entry_index].position_id=position_id;
            active_trades[entry_index].ticket=trans.position;
            active_trades[entry_index].order_ticket=order_ticket;
            active_trades[entry_index].entry_deal_ticket=trans.deal;
            if(active_trades[entry_index].decision_id=="") active_trades[entry_index].decision_id=DecisionIdFromComment(deal_comment,position_id);
        }
        return;
    }
    if(entry_type!=DEAL_ENTRY_OUT && entry_type!=DEAL_ENTRY_OUT_BY)return;

    int live_index=FindTradeByIdentity(0,0,position_id,deal_comment); int closed_index=-1; bool found=live_index>=0;
    TradeInfo record;
    if(found) record=active_trades[live_index];
    else { closed_index=FindClosedTradeByIdentity(position_id,trans.position); found=closed_index>=0; if(found) record=closed_trades[closed_index]; }
    if(found && record.outcome_written) return;

    if(!found) {
        if(runtime_debug_logs) Print("Outcome deferred: no tracked trade decision=",DecisionIdFromComment(deal_comment,position_id)," position=",position_id," deal=",trans.deal);
        return;
    }

    datetime exit_time=(datetime)HistoryDealGetInteger(trans.deal,DEAL_TIME);
    double exit_price=HistoryDealGetDouble(trans.deal,DEAL_PRICE);
    double gross=HistoryDealGetDouble(trans.deal,DEAL_PROFIT);
    double commission=HistoryDealGetDouble(trans.deal,DEAL_COMMISSION);
    double swap=HistoryDealGetDouble(trans.deal,DEAL_SWAP);
    double deal_volume=HistoryDealGetDouble(trans.deal,DEAL_VOLUME);
    bool valid_deal=exit_time>0 && exit_price>0 && deal_volume>0 && MathIsValidNumber(gross) && MathIsValidNumber(commission) && MathIsValidNumber(swap);
    if(!valid_deal || record.entry_time<=0 || exit_time<record.entry_time || record.entry_price<=0) {
        if(runtime_debug_logs) Print("Outcome deferred: invalid deal data decision=",record.decision_id," position=",position_id," deal=",trans.deal," time=",(long)exit_time," price=",DoubleToString(exit_price,_Digits)," volume=",DoubleToString(deal_volume,2));
        return;
    }

    string id=record.decision_id!=""?record.decision_id:DecisionIdFromComment(deal_comment,position_id);
    double entry_price=record.entry_price;
    double fill_price=entry_price;
    if(HistorySelectByPosition(position_id))
        for(int d=0;d<HistoryDealsTotal();d++){ ulong dt=HistoryDealGetTicket(d); if(dt>0 && HistoryDealGetInteger(dt,DEAL_ENTRY)==DEAL_ENTRY_IN){ fill_price=HistoryDealGetDouble(dt,DEAL_PRICE); break; } }

    double net=gross+commission+swap;

    if(found) {
        record.cumulative_gross+=gross; record.cumulative_commission+=commission; record.cumulative_swap+=swap; record.cumulative_net+=net;
        double deal_move=record.trade_type==0?HistoryDealGetDouble(trans.deal,DEAL_PRICE)-fill_price:fill_price-HistoryDealGetDouble(trans.deal,DEAL_PRICE);
        record.realized_volume+=deal_volume; record.cumulative_realized_move+=deal_move*deal_volume; record.last_exit_time=exit_time;
        record.final_closing_reason=EnumToString((ENUM_DEAL_REASON)HistoryDealGetInteger(trans.deal,DEAL_REASON));
        if(record.partial_info=="NONE" || record.partial_info=="RECOVERED_AFTER_RESTART") record.partial_info="";
        if(record.partial_info!="") record.partial_info+="|";
        record.partial_info+=StringFormat("DEAL_VOLUME=%.2f;GROSS=%.2f;COMMISSION=%.2f;SWAP=%.2f;NET=%.2f;REASON=%s",deal_volume,gross,commission,swap,net,record.final_closing_reason);
        if(live_index>=0) active_trades[live_index]=record; else if(closed_index>=0) closed_trades[closed_index]=record;
        if(IsPositionStillOpen(position_id,trans.position)) return;
        if(record.outcome_written) return;
    }

    //--- R multiple from price distance: (realised move) / (initial risk distance)
    double risk_distance=found?MathAbs(record.entry_price-record.initial_stop_loss):0;
    double move=found?((record.trade_type==0)?(exit_price-fill_price):(fill_price-exit_price)):0;
    double realized_move=found && record.original_lot_size>0?record.cumulative_realized_move/record.original_lot_size:move;
    double r_multiple=(risk_distance>0)?realized_move/risk_distance:0;

    double mfe_points=found?record.mfe/_Point:0;
    double mae_points=found?record.mae/_Point:0;
    string partial_info=found?record.partial_info:"UNLINKED";
    if(found) partial_info+=StringFormat("|TP1=%s;TP2=%s;TP3=%s;REALIZED_VOLUME=%.2f;REMAINING=%.2f",record.tp1_hit?"1":"0",record.tp2_hit?"1":"0",record.tp3_hit?"1":"0",record.realized_volume,record.remaining_lot_size);

    record.outcome_written=true;
    if(live_index>=0) active_trades[live_index]=record; else if(closed_index>=0) closed_trades[closed_index]=record;
    if(!WriteOutcomeRecord(id,position_id,entry_price,fill_price,exit_price,exit_time,
                       found?record.cumulative_gross:gross,found?record.cumulative_commission:commission,found?record.cumulative_swap:swap,found?record.cumulative_net:net,
                       (found?record.cumulative_net:net)>=0?"WIN":"LOSS",r_multiple,mfe_points,mae_points,
                       record.final_closing_reason,partial_info,record,true)) {
        record.outcome_written=false;
        if(live_index>=0) active_trades[live_index]=record; else if(closed_index>=0) closed_trades[closed_index]=record;
        if(runtime_debug_logs) Print("Outcome deferred: outcome write failed decision=",id," position=",position_id," deal=",trans.deal);
        return;
    }

    if(live_index>=0) {
        active_trades[live_index]=record;
        if(!position.SelectByTicket(active_trades[live_index].ticket)) RemoveTradeFromArray(live_index);
    }
}
